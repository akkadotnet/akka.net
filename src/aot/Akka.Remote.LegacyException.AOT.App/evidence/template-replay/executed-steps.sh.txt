#!/usr/bin/env bash
set -euo pipefail
rm -rf src/aot/Akka.Remote.LegacyException.AOT.App/bin src/aot/Akka.Remote.LegacyException.AOT.App/obj
mkdir -p "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof"
dotnet --info > "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/sdk-info.log"
dotnet publish src/aot/Akka.Remote.LegacyException.AOT.App -c Release -r linux-x64 \
  -o "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-publish" 2>&1 | tee "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/publish.log"

export EXPECTED_MARKER='[legacy-exceptions-native] OK reads=114 rejected=15 overrides=3 empty-sentinels=3'
set -euo pipefail
timeout 300 "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-publish/Akka.Remote.LegacyException.AOT.App" \
  2>&1 | tee "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/run.log"
grep -Fxq "$EXPECTED_MARKER" \
  "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/run.log"

set -euo pipefail
dotnet run scripts/CheckAotWarnings.cs -- \
  --log "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/publish.log" \
  --baseline src/aot/Shared/remote-aot-warnings.baseline.txt \
  --scope "src/core/Akka.Remote/,src/core/Akka.Serialization.V2/,src/aot/Akka.Remote.LegacyException.AOT.App/,src/aot/Shared/" \
  --repo-root "/tmp/akka-serialization-v2/native-legacy-exceptions" \
  2>&1 | tee "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/warnings.log"
grep -Eq '^  no source location: 0 warning' "/tmp/akka-serialization-v2/legacy-template-replay/RemoteLegacyExceptionAotCanary-proof/warnings.log"
